### 1.2 `access.ts`

```ts
export type ContentAccess = {
  canEdit: boolean   // manager (admin, dept_admin of the course's department) or instructor/TA of the section
  canRead: boolean   // canEdit or enrolled (myRole === 'student')
  canGrades: boolean // gradebookAccess(...).view
  ownRole?: 'instructor' | 'ta' | 'student'
}

/** Display only; the server authorises every request (spec 6 §4). */
export function contentAccess(m: MyMembership, userId: string, section: Section, myRole?: 'instructor' | 'ta' | 'student'): ContentAccess {
  const { manager, staffRole } = sectionAccess(m, userId, section)
  const canEdit = manager || !!staffRole
  return {
    canEdit,
    canRead: canEdit || myRole === 'student',
    canGrades: gradebookAccess(m, userId, section).view,
    ownRole: staffRole ?? myRole,
  }
}

export function useContentAccess(slug: string, section: Section): ContentAccess {
  const { membership } = useUniversity()
  const me = useMe().data
  const myRole = useMySections(slug).data?.terms.flatMap((g) => g.sections).find((s) => s.id === section.id)?.myRole
  return contentAccess(membership, me?.user.id ?? '', section, myRole)
}

export const authorName = (a: { displayName: string; email?: string }) => a.displayName || a.email || 'Unnamed member'
```

`authorName` accepts an optional `email` so it keeps working unchanged if escalation §12 item 1 adds it to the contract.

While `/me/sections` is loading, an enrolled student's `canRead` is false and the strip appears when it arrives (students normally arrive from My sections, which has filled that cache).

### 1.3 `queries.ts`

```ts
export const sectionKey = (slug: string, id: string) => [slug, 'sections', 'detail', id] as const
export const announcementsKey = (slug: string, id: string) => [slug, 'sections', id, 'announcements'] as const
export const weeksKey = (slug: string, id: string) => [slug, 'sections', id, 'weeks'] as const // L2b

export function useAnnouncements(slug: string, sectionId: string) {
  return useCursorList<Announcement>(announcementsKey(slug, sectionId), (cursor) =>
    unwrap(api.GET('/universities/{slug}/sections/{id}/announcements', {
      params: { path: { slug, id: sectionId }, query: { limit: 20, cursor } },
    })),
  )
}

/** Invalidates one content key and the audit log; await it before showing a status line. */
export function useContentRefresh(slug: string) {
  const qc = useQueryClient()
  return (key: readonly unknown[]) =>
    Promise.all([key, [slug, 'audit-log']].map((queryKey) => qc.invalidateQueries({ queryKey })))
}

// L2b
export function useWeeks(slug: string, sectionId: string): UseQueryResult<S['WeekList'], ApiError> // useQuery, key weeksKey, GET .../weeks

// fetch sets the multipart boundary itself, so no Content-Type (same as api/upload.ts)
export function uploadMaterial(slug: string, sectionId: string, weekId: string, file: File): Promise<Material> {
  const form = new FormData()
  form.append('file', file)
  return unwrap(api.POST('/universities/{slug}/sections/{id}/weeks/{weekId}/materials', {
    params: { path: { slug, id: sectionId, weekId } },
    body: form as never,
    bodySerializer: (b: unknown) => b as FormData,
  }))
}
```

No `title` part is sent: the server defaults the title to the file name.

| Write | Request | Invalidate (then status) |
|---|---|---|
| Save description | `PUT .../sections/{id}/description` `{ description }` | `sectionKey` |
| Post / edit / delete announcement | `POST .../announcements` `{ title, body, email }`; `PATCH .../announcements/{aid}` with changed fields only; `DELETE` | `announcementsKey` |
| Add / edit week | `POST .../weeks` `{ title, notes }`; `PATCH .../weeks/{wid}` changed fields only | `weeksKey` |
| Move week | `PATCH .../weeks/{wid}` `{ position: week.position - 1 }` or `+ 1` | `weeksKey` |
| Delete week | `DELETE .../weeks/{wid}` | `weeksKey` |
| Upload batch | `POST .../weeks/{wid}/materials` per file | `weeksKey` once, after the last file |
| Rename / move / delete file | `PATCH .../materials/{mid}` `{ title }` or `{ weekId }`; `DELETE` | `weeksKey` |

Every refresh also invalidates `[slug, 'audit-log']` (`useContentRefresh`). Existing writes elsewhere that invalidate `[slug, 'sections']` (staff editor, course and section dialogs) also refetch these content keys because they share the prefix; harmless.

A PATCH with no changed field closes the dialog without a request (as `course-dialog.tsx` does).

### 1.4 `plain-text.tsx`

```tsx
const url = /(https?:\/\/[^\s<>"]*[^\s<>".,:;!?)\]'])/g

/** Plain text with line breaks kept; http(s) URLs become links. Nothing else is interpreted. */
export function PlainText({ text, className = '' }: { text: string; className?: string }) {
  return (
    <div className={`whitespace-pre-wrap wrap-break-word ${className}`}>
      {text.split(url).map((part, i) =>
        i % 2 === 1 ? (
          <a key={i} href={part} target="_blank" rel="noopener noreferrer" className="text-accent underline hover:text-accent-hover">
            {part}
          </a>
        ) : (
          part
        ),
      )}
    </div>
  )
}

const n = (v: number) => v.toLocaleString('en-US')

/** "120 / 10,000 characters", or "10,012 / 10,000 characters, 12 too many". Code points, as the server counts. */
export function countHint(value: string, max: number): string {
  const c = [...value].length
  return `${n(c)} / ${n(max)} characters${c > max ? `, ${n(c - max)} too many` : ''}`
}
```

`split` with one capture group puts the URLs at odd indexes. A trailing period or closing bracket stays outside the link.

### 5.4 `files.ts`

```ts
export const MAX_BYTES = 104_857_600
export const ALLOWED = ['pdf', 'png', 'jpg', 'jpeg', 'gif', 'webp', 'zip', 'ppt', 'pptx', 'doc', 'docx', 'xls', 'xlsx', 'txt', 'md', 'csv'] as const
export const ACCEPT = ALLOWED.map((e) => `.${e}`).join(',')

/** Text after the last dot, lower case; '' when there is none (as the server reads it, spec 6 §6). */
export const extension = (name: string) => { const i = name.lastIndexOf('.'); return i < 0 ? '' : name.slice(i + 1).toLowerCase() }

export function formatSize(n: number): string {
  if (n < 1024) return `${n} B`
  const [v, unit] = n < 1024 ** 2 ? [n / 1024, 'KB'] : [n / 1024 ** 2, 'MB']
  return `${v < 10 ? Number(v.toFixed(1)) : Math.round(v)} ${unit}`
}

export function fileIcon(contentType: string): LucideIcon {
  if (contentType.startsWith('image/')) return ImageIcon
  if (contentType === 'application/zip') return FileArchive
  if (contentType.includes('powerpoint') || contentType.includes('presentationml')) return Presentation
  if (contentType === 'application/pdf' || contentType.includes('msword') || contentType.includes('wordprocessingml') || contentType.startsWith('text/plain')) return FileText
  return FileIcon
}
```

Import the aliases `ImageIcon` and `FileIcon` from `lucide-react` (the plain names shadow the DOM's `Image` and `File`). `ALLOWED` mirrors the server's map (P10-L1 §5.6); the server stays the authority.

### 5.5 `upload.ts`

```ts
export type Failed = { name: string; reason: string }

/** Reason a file is refused before sending, or undefined. Order: type, empty, size. */
export function precheck(f: File): string | undefined {
  if (!(ALLOWED as readonly string[]).includes(extension(f.name))) return "This file type can't be uploaded."
  if (f.size === 0) return 'The file is empty.'
  if (f.size > MAX_BYTES) return 'It is larger than 100 MB.'
}

/** Sends the files that pass precheck one at a time, in order; a failure never stops the rest. */
export async function runUploads(
  files: File[],
  send: (f: File) => Promise<unknown>,
  onProgress: (line: string) => void,
): Promise<{ done: number; total: number; failed: Failed[] }> {
  const failed: Failed[] = []
  const ok: File[] = []
  for (const f of files) {
    const r = precheck(f)
    if (r) failed.push({ name: f.name, reason: r })
    else ok.push(f)
  }
  let done = 0
  for (const [i, f] of ok.entries()) {
    onProgress(`Uploading ${i + 1} of ${ok.length}: ${f.name}`)
    try { await send(f); done++ } catch (e) { failed.push({ name: f.name, reason: uploadReason(e) }) }
  }
  return { done, total: files.length, failed }
}

export function uploadReason(e: unknown): string {
  if (!(e instanceof ApiError)) return 'The upload was interrupted. Check your connection and try again.'
  const d = e.details[0]
  if (e.code === 'validation_failed' && d) {
    if (d.field === 'materials') return 'This week already has 200 files.'
    if (d.message.includes('larger than 100 MB')) return 'It is larger than 100 MB.'
    if (d.message === 'is empty') return 'The file is empty.'
    if (d.message.startsWith('file type not allowed')) return "This file type can't be uploaded."
    return sentence(d.message)
  }
  return errorMessage(e)
}

export function finalLine(r: { done: number; total: number }, week: string): string {
  if (r.done === r.total) return `Uploaded ${r.done} ${r.done === 1 ? 'file' : 'files'} to ${week}.`
  if (r.done === 0) return `No files were uploaded to ${week}.`
  return `Uploaded ${r.done} of ${r.total} files to ${week}.`
}
```

Page side (`MaterialsPage`): `onUpload(week, files)` sets `upload = { weekId, line: '', failed: [], busy: true }`, runs `runUploads(files, (f) => uploadMaterial(slug, sectionId, week.id, f), (line) => setUpload(u => ({ ...u, line })))`, then `await refresh(weeksKey(...))`, then sets `line = finalLine(r, week.title)`, `failed = r.failed`, `busy = false`. The error banner title is `` `${k} ${k === 1 ? 'file' : 'files'} not uploaded` ``, its body `<ul className="list-disc pl-5">` of `<li><span className="font-bold">{name}</span>: {reason}</li>`. A new batch in any week clears the previous result. While `busy`, a `useEffect` adds a `beforeunload` listener that calls `e.preventDefault()` and removes it when `busy` ends.

## 7. Accessibility

- Landmarks unchanged; the strip is a `<nav aria-label="Section pages">`; the current tab has `aria-current="page"` (TanStack `Link`).
- Headings: one `h1` per page (the band), `h2` per tab content ("About this section", "Announcements", "Materials"), `h3` per announcement and per week. Lists are `ul`/`ol` named by their `h2` or `h3`.
- Every control has a visible label; repeated controls carry a full accessible name that starts with the visible text (WCAG 2.5.3): "Edit {title}", "Download {title}", "Move {week} up". Icon-only controls (the two move arrows) carry `aria-label` and `title`. Every Lucide icon next to text is `aria-hidden="true"`; file type icons too (the type is in the file name).
- Dialogs: native `<dialog>` traps focus and closes on Escape; the first focusable is Cancel in confirms; focus returns to the trigger, or to the tab's `h2` (`tabIndex={-1}`) after a delete removed the trigger.
- Live regions: `StatusMessage` (`role="status"`) per page for saves and deletes; the week's upload line (`role="status"`, always present for editors); upload failures in an `ErrorBanner` (`role="alert"`, announced once per batch); field errors via `FormField` (`role="alert"`) and `useFocusFirstError`.
- No state by colour alone: "Emailed" and "Edited" are words; the danger style is only on Delete buttons and error banners.
- Links in running text (inside `PlainText`) are underlined; the Download link is underlined.
